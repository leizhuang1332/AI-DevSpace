1// 「占满父 flex 容器高度 + 内滚动」的统一封装
2// fill=true 时高度 100% (依赖父提供 definite height)
3'use client'
4import { type ReactNode } from 'react'
5import { cn } from '@/lib/cn'
6
7export interface ScrollContainerProps {
8 children: ReactNode
9 fill?: boolean // 父级有约束高度 → h-full
10 maxHeight?: string // 无约束父级 → 兜底硬上限
11 className?: string
12 'data-testid'?: string
13}
14
15export function ScrollContainer({
16 children, fill, maxHeight, className, ...rest
17}: ScrollContainerProps) {
18 return (
19 <div
20 className={cn(
21 'overflow-auto',
22 fill && 'h-full min-h-0',
23 !fill && maxHeight && `max-h-[${maxHeight}]`,
24 className,
25 )}
26 {...rest}
27 >{children}</div>
28 )
29}

关键点 · 风险 · 验收

  • i抽象的价值
    同一类 CSS 模式("占满父高度 + 内滚动")在项目里已用 3+ 次(IssueList / RunLog / DocumentReaderPane),抽出 ScrollContainer 收口,以后写新 scrollable 区不用重抄 className 链。
  • !本次单独引入是 over-engineering
    这是为单点 bug 引入 abstraction。若短期内(下个 ticket)还有 2+ 个新 scrollable 区,D 划算;若就这一处,A 更轻。
  • ifill / maxHeight 二选一
    组件 API:fill 时拿父容器高度(同 A),无 fill 时给 maxHeight 兜底(同 C)。Caller 不用纠结 h-full vs max-h,语义更显。
  • !需要写 ScrollContainer 自己的测试
    fill=true 父级高度为 0 时不能让内容溢出导致页面拉长;maxHeight 模式下也应保证视觉一致。新组件 = 新测试 = 新 review surface。
  • !建议 ADR 写 scroll 契约
    抽象本身是契约变更:scrollable 区域统一由 ScrollContainer 提供,组件不再各自手写 flex-1 overflow-auto。reviewer 看 git blame 时要交代清楚。
  • ✓顺手把 RunLog / ReaderPane 也迁过来
    同一个 ticket 内完成:ScrollContainer 上线 → AnalysisIssueList / AnalysisRunLogPanel / DocumentReaderPane 全部采用,避免「新 abstraction 半途而废」。
